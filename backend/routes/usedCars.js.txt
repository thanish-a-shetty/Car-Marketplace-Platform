// backend/routes/usedCars.js
const express = require('express');
const multer = require('multer');
const UsedCar = require('../models/UsedCar');
const jwt = require('jsonwebtoken');
const path = require('path');

const router = express.Router();

// Multer setup for image upload
const storage = multer.diskStorage({
  destination: function (req, file, cb) {
    cb(null, path.join(__dirname, '../uploads'));
  },
  filename: function (req, file, cb) {
    cb(null, Date.now() + '-' + file.originalname);
  }
});
const upload = multer({ storage });

// Middleware to verify JWT
function auth(req, res, next) {
  const token = req.header('Authorization')?.replace('Bearer ', '');
  if (!token) return res.status(401).json({ message: 'No token, authorization denied' });
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded.userId;
    next();
  } catch (err) {
    res.status(401).json({ message: 'Token is not valid' });
  }
}

// Get all used cars
router.get('/', async (req, res) => {
  const cars = await UsedCar.find().populate('seller', 'email');
  res.json(cars);
});

// Add a used car (requires auth)
router.post('/', auth, upload.single('image'), async (req, res) => {
  try {
    const { model, year, price, location, kmRun, mileage } = req.body;
    const car = new UsedCar({
      model,
      year,
      price,
      location,
      kmRun,
      mileage,
      image: req.file ? `/uploads/${req.file.filename}` : undefined,
      seller: req.user
    });
    await car.save();
    res.status(201).json(car);
  } catch (err) {
    res.status(500).json({ message: 'Server error' });
  }
});

module.exports = router;
